52void pooled_connection::release() noexcept
53{
54 if (pool_ && backend_) {
55 // Return to pool (noexcept contract: the pool's return_connection
56 // must not throw; it only moves pointers and notifies the CV).
57 pool_->return_connection(std::move(backend_), broken_);
58 }
59 pool_.reset();
60 backend_.reset();
61 broken_ = false;
62}

84std::shared_ptr<connection_pool> connection_pool::create(
85 pool_config config,
86 connection_factory factory,
87 connection_validator validator)
88{
89 if (config.max_size == 0) {
90 config.max_size = 1;
91 }
92 if (config.min_size > config.max_size) {
93 config.min_size = config.max_size;
94 }
95
96 if (!validator) {
97 validator = default_validator();
98 }
99
100 // Use new because the ctor is private; std::make_shared would need a friend.
101 auto pool = std::shared_ptr<connection_pool>(new connection_pool(
102 std::move(config), std::move(factory), std::move(validator)));
103
104 // Pre-warm up to min_size. Failures are tolerated: the pool will retry
105 // on the first acquire() call.
106 {
107 std::unique_lock<std::mutex> lock(pool->mutex_);
108 while (pool->total_connections_ < pool->config_.min_size) {
109 auto backend = pool->create_locked(lock);
110 if (!backend) {
111 break; // Abort pre-warming; acquire() will retry later.
112 }
113 pool->idle_.push_back(std::move(backend));
114 }
115 }
116
117 return pool;
118}

134void connection_pool::shutdown()
135{
136 std::deque<std::unique_ptr<database_backend>> drained;
137 {
138 std::lock_guard<std::mutex> lock(mutex_);
139 if (shut_down_) {
140 return;
141 }
142 shut_down_ = true;
143 drained.swap(idle_);
144 total_connections_ -= drained.size();
145 }
146 // Destroy idle connections outside the lock. shutdown() on each backend
147 // is invoked via the backend destructor.
148 drained.clear();
149 cv_.notify_all();
150}

178Result<pooled_connection> connection_pool::acquire(std::chrono::milliseconds timeout)
179{
180 const auto deadline = std::chrono::steady_clock::now() + timeout;
181
182 std::unique_lock<std::mutex> lock(mutex_);
183
184 for (;;) {
185 if (shut_down_) {
186 return make_pool_error(error_code::invalid_state,
187 "connection_pool has been shut down");
188 }
189
190 // Prefer an idle connection.
191 if (auto backend = take_idle_locked()) {
192 // Validate (with the lock released) before handing out.
193 if (config_.validate_on_acquire) {
194 lock.unlock();
195 const bool ok = validator_(*backend);
196 lock.lock();
197 if (!ok) {
198 // Discard and try to create a replacement.
199 --total_connections_;
200 ++replaced_connections_;
201 backend.reset();
202 auto fresh = create_locked(lock);
203 if (!fresh) {
204 // Creation failed; loop to wait again or timeout.
205 } else {
206 ++active_connections_;
207 ++total_acquires_;
208 return pooled_connection(shared_from_this(), std::move(fresh));
209 }
210 } else {
211 ++active_connections_;
212 ++total_acquires_;
213 return pooled_connection(shared_from_this(), std::move(backend));
214 }
215 } else {
216 ++active_connections_;
217 ++total_acquires_;
218 return pooled_connection(shared_from_this(), std::move(backend));
219 }
220 }
221
222 // No idle connection. Can we create a new one?
223 if (total_connections_ < config_.max_size) {
224 auto fresh = create_locked(lock);
225 if (fresh) {
226 ++active_connections_;
227 ++total_acquires_;
228 return pooled_connection(shared_from_this(), std::move(fresh));
229 }
230 // Creation failed; fall through to waiting (so we back off a bit).
231 }
232
233 // Pool is saturated (or creation failed): wait for a release.
234 ++waiters_;
235 const auto wait_result = cv_.wait_until(lock, deadline);
236 --waiters_;
237
238 if (wait_result == std::cv_status::timeout &&
239 std::chrono::steady_clock::now() >= deadline) {
240 ++acquire_timeouts_;
241 return make_pool_error(error_code::timeout,
242 "timed out waiting for a pooled connection");
243 }
244 // Otherwise loop and try again.
245 }
246}

248void connection_pool::return_connection(std::unique_ptr<database_backend> backend, bool broken)
249{
250 if (!backend) {
251 return;
252 }
253
254 {
255 std::lock_guard<std::mutex> lock(mutex_);
256
257 // Invariant: only decrement active when we were counted as active.
258 if (active_connections_ > 0) {
259 --active_connections_;
260 }
261
262 if (broken || shut_down_) {
263 // Drop the backend entirely.
264 if (total_connections_ > 0) {
265 --total_connections_;
266 }
267 if (broken) {
268 ++replaced_connections_;
269 }
270 } else {
271 // Return to idle (LIFO keeps the most-recently-used connection hot).
272 idle_.push_back(std::move(backend));
273 }
274 }
275
276 cv_.notify_one();
277}

279std::unique_ptr<database_backend> connection_pool::create_locked(std::unique_lock<std::mutex>& lock)
280{
281 // Reserve capacity while holding the lock. Without this reservation, every
282 // waiter can observe the same available slot after the factory call unlocks
283 // the mutex, allowing the pool to grow beyond max_size.
284 ++total_connections_;
285
286 // Factory may block on network I/O; drop the lock while calling it.
287 lock.unlock();
288 std::unique_ptr<database_backend> backend;
289 try {
290 backend = factory_();
291 } catch (...) {
292 backend.reset();
293 }
294 lock.lock();
295
296 if (!backend) {
297 --total_connections_;
298 ++failed_creations_;
299 return nullptr;
300 }
301
302 return backend;
303}

305std::unique_ptr<database_backend> connection_pool::take_idle_locked()
306{
307 if (idle_.empty()) {
308 return nullptr;
309 }
310 // LIFO: reuse the most recently released connection first.
311 auto backend = std::move(idle_.back());
312 idle_.pop_back();
313 return backend;
314}
