22std::unique_ptr<backends::thread_backend> thread_adapter::create_backend(
23 const db_thread_config& config,
24 thread_backend_type backend_type)
25{
26 switch (backend_type)
27 {
28 case thread_backend_type::auto_select:
29 case thread_backend_type::fallback:
30 return std::make_unique<backends::fallback_thread_backend>(config);
31
32 case thread_backend_type::null:
33 return std::make_unique<backends::null_thread_backend>(config);
34
35 default:
36 return std::make_unique<backends::fallback_thread_backend>(config);
37 }
38}
