37struct lifecycle_observer
38{
39 int shutdown_call_count = 0;
40 int partial_cleanup_count = 0;
41 bool member_alive = false;
42 bool member_alive_during_shutdown = false;
43 bool partial_resource_alive = false;
44};

46class observed_member
47{
48public:
49 explicit observed_member(lifecycle_observer* observer) noexcept
50 : observer_(observer)
51 {
52 if (observer_) {
53 observer_->member_alive = true;
54 }
55 }
56
57 ~observed_member()
58 {
59 if (observer_) {
60 observer_->member_alive = false;
61 }
62 }
63
64private:
65 lifecycle_observer* observer_;
66};

68class tracked_resource
69{
70public:
71 explicit tracked_resource(lifecycle_observer* observer) noexcept
72 : observer_(observer)
73 {
74 if (observer_) {
75 observer_->partial_resource_alive = true;
76 }
77 }
78
79 ~tracked_resource()
