31// Example Entity: User
32class User : public entity_base
33{
34 ENTITY_TABLE("users")
35
36 // Define fields with constraints
37 ENTITY_FIELD(int64_t, id, primary_key() | auto_increment())
38 ENTITY_FIELD(std::string, username, not_null() | unique() | index("idx_username"))
39 ENTITY_FIELD(std::string, email, not_null() | unique())
40 ENTITY_FIELD(std::string, full_name, not_null())
41 ENTITY_FIELD(std::chrono::system_clock::time_point, created_at, default_now())
42 ENTITY_FIELD(bool, is_active, not_null())
43
44 ENTITY_METADATA()
45
46public:
47 User() {
48 // Set default values
49 is_active = true;
50 created_at = std::chrono::system_clock::now();
51 }
52
53 // Custom validation method
54 bool is_valid() const {
55 return !username.get().empty() &&
56 !email.get().empty() &&
57 email.get().find('@') != std::string::npos;
58 }
59
60 // Display method
61 void print_info() const {
62 std::cout << "User ID: " << id.get()
63 << ", Username: " << username.get()
64 << ", Email: " << email.get()
65 << ", Active: " << (is_active.get() ? "Yes" : "No") << std::endl;
66 }
67};

73 metadata_.add_field(email_field());
74 metadata_.add_field(full_name_field());
