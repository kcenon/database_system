36 bool database_manager::set_mode(const database_types& database_type)
37 {
38 if (connected_)
39 {
40 return false;
41 }
42
43 database_.reset();
44
45 // Create the appropriate backend directly
46 // This ensures backends are linked even when static library registration
47 // might not work due to initialization order
48 switch (database_type)
49 {
50 case database_types::postgres:
51 database_ = backends::postgresql_backend::create();
52 break;
53 case database_types::sqlite:
54 database_ = backends::sqlite_backend::create();
55 break;
56#ifdef USE_MONGODB
57 case database_types::mongodb:
58 database_ = backends::mongodb_backend::create();
59 break;
60#endif
61#ifdef USE_REDIS
62 case database_types::redis:
63 database_ = backends::redis_backend::create();
64 break;
65#endif
66 default:
67 return false;
68 }
69
70 if (database_ == nullptr)
71 {
72 return false;
73 }
74
75 return true;
76 }

88 kcenon::common::VoidResult database_manager::connect_result(const std::string& connect_string)
89 {
90 if (!database_)
91 {
92 return kcenon::common::VoidResult(
93 kcenon::common::error_info{static_cast<int>(error_code::invalid_state), "No database backend configured", "database_manager"});
94 }
95
96 // Store connection string for potential reconnection
97 connect_string_ = connect_string;
98
99 // Use database_backend's initialize method with connection_config
100 auto config = core::connection_config::from_string(connect_string);
101 auto result = database_->initialize(config);
102
103 if (result.is_ok())
104 {
105 connected_ = true;
106 }
107 return result;
108 }
