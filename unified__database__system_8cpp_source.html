58static std::string backend_type_to_name(backend_type type) {
59 switch (type) {
60 case backend_type::postgres: return "postgresql";
61 case backend_type::sqlite: return "sqlite";
62 case backend_type::mongodb: return "mongodb";

74static std::shared_ptr<core::database_backend> create_backend(backend_type type) {
75 switch (type) {
76 case backend_type::postgres:
77 return std::make_shared<postgres_manager>();
78 default: {
79 // Fall back to backend_registry for other types
80 auto name = backend_type_to_name(type);
81 if (!name.empty()) {
82 auto backend = core::backend_registry::instance().create(name);
83 if (backend) {
84 return std::shared_ptr<core::database_backend>(std::move(backend));
85 }
86 }
87 return nullptr;
88 }

95static std::string value_to_string(const core::database_value& value) {
96 return std::visit([](const auto& v) -> std::string {
97 using T = std::decay_t<decltype(v)>;
98 if constexpr (std::is_same_v<T, std::string>) {
99 return v;
100 } else if constexpr (std::is_same_v<T, int64_t>) {
101 return std::to_string(v);
102 } else if constexpr (std::is_same_v<T, double>) {
103 return std::to_string(v);
104 } else if constexpr (std::is_same_v<T, bool>) {
105 return v ? "true" : "false";
106 } else if constexpr (std::is_same_v<T, std::nullptr_t>) {
107 return "NULL";
108 }
109 return "";
110 }, value);
111}

116static query_result convert_result(
117 const core::database_result& db_result,
118 std::chrono::microseconds exec_time) {
119
120 query_result result;
121 result.rows.reserve(db_result.size());
122
123 // Convert each row: map<string, database_value> -> map<string, string>
124 for (const auto& db_row : db_result) {
125 row_data converted_row;
126 for (const auto& [key, value] : db_row) {
127 converted_row[key] = value_to_string(value);
128 }
129 result.rows.push_back(std::move(converted_row));
130 }
131
132 result.affected_rows = db_result.size();
133 result.execution_time = exec_time;
134
135 return result;
136}

147std::vector<core::database_value> convert_params(const std::vector<query_param>& params) {
148 std::vector<core::database_value> values;
149 values.reserve(params.size());
150 for (const auto& p : params) {
151 if (p.is_null()) {
152 values.emplace_back(nullptr);
153 } else {
154 values.emplace_back(p.get_value());
155 }
156 }
157 return values;
158}

163bool is_select_query(const std::string& query) {
164 // Skip leading whitespace and find the first keyword
165 auto pos = query.find_first_not_of(" \t\n\r");
166 if (pos == std::string::npos) return false;
167 // Case-insensitive check for SELECT (or WITH which precedes CTEs)
168 auto keyword = query.substr(pos, 6);
169 for (auto& c : keyword) c = static_cast<char>(std::tolower(static_cast<unsigned char>(c)));
170 return keyword.substr(0, 6) == "select" || keyword.substr(0, 4) == "with";
171}

179class transaction_impl : public transaction {
180public:
181 explicit transaction_impl(std::shared_ptr<core::database_backend> backend)
182 : backend_(std::move(backend)), active_(false) {
183
184 // Begin transaction
185 if (backend_) {
186 auto result = backend_->begin_transaction();
187 if (result.is_ok()) {
188 active_ = true;
189 }
190 }
191 }
192
193 ~transaction_impl() override {
194 if (active_) {
195 // Auto-rollback if not committed
196 backend_->rollback_transaction();
197 }
198 }
199
200 kcenon::common::Result<query_result> execute(
201 const std::string& query,
202 const std::vector<query_param>& params) override {
203
204 if (!active_) {
205 return make_error_result<query_result>("Transaction not active", static_cast<int>(error_code::invalid_state), "transaction");
206 }
207
208 auto start = std::chrono::steady_clock::now();
209
210 kcenon::common::Result<core::database_result> db_result = [&]() {
211 if (params.empty()) {
212 if (is_select_query(query)) {
213 return backend_->select_query(query);
214 }
215 auto exec_res = backend_->execute_query(query);
216 if (exec_res.is_err()) {
217 return kcenon::common::Result<core::database_result>(exec_res.error());
218 }
219 return kcenon::common::Result<core::database_result>(core::database_result{});
220 }
221 auto values = convert_params(params);
222 if (is_select_query(query)) {
223 return backend_->select_prepared(query, values);
224 }
225 auto exec_res = backend_->execute_prepared(query, values);
226 if (exec_res.is_err()) {
227 return kcenon::common::Result<core::database_result>(exec_res.error());
228 }
229 return kcenon::common::Result<core::database_result>(core::database_result{});
230 }();
231
232 auto duration = std::chrono::duration_cast<std::chrono::microseconds>(
233 std::chrono::steady_clock::now() - start);
234
235 if (db_result.is_err()) {
236 return db_result.error();
237 }
238
239 auto qr = convert_result(db_result.value(), duration);
240 // execute_query / execute_prepared return VoidResult (no row count),
241 // so affected_rows stays 0 after convert_result. For successful DML
242 // queries we report at least 1 affected row.
243 if (!is_select_query(query) && qr.affected_rows == 0) {
244 qr.affected_rows = 1;
245 }
246 return qr;
247 }
248
249 kcenon::common::VoidResult commit() override {
250 if (!active_) {
251 return make_error("Transaction not active", static_cast<int>(error_code::invalid_state), "transaction");
252 }
253
254 auto result = backend_->commit_transaction();
255 if (result.is_err()) {
256 return make_error("Commit failed: " + result.error().message, static_cast<int>(error_code::invalid_state), "transaction");
257 }
258
259 active_ = false;
260 return kcenon::common::ok();
261 }
262
263 kcenon::common::VoidResult rollback() override {
264 if (!active_) {
265 return make_error("Transaction not active", static_cast<int>(error_code::invalid_state), "transaction");
266 }
267
268 auto result = backend_->rollback_transaction();
269 if (result.is_err()) {
270 return make_error("Rollback failed: " + result.error().message, static_cast<int>(error_code::invalid_state), "transaction");
271 }
272
273 active_ = false;
274 return kcenon::common::ok();
275 }
276
277 bool is_active() const override {
278 return active_;
279 }
280
281private:
282 std::shared_ptr<core::database_backend> backend_;
283 bool active_;
284};

310
311 kcenon::common::VoidResult connect(backend_type backend, const std::string& connection_string) {
312 std::lock_guard<std::mutex> lock(mutex_);
313
314 auto init_result = ensure_initialized();
315 if (!init_result.is_ok()) {
316 return init_result;
317 }
318
319 if (connected_) {
320 return make_error("Already connected", static_cast<int>(error_code::invalid_state), "unified_database_system");
321 }
322
323 backend_type_ = backend;
324 connection_string_ = connection_string;
325
326 // Create backend instance
327 backend_ = create_backend(backend);
328 if (!backend_) {
329 return make_error("Unsupported backend type", static_cast<int>(error_code::invalid_argument), "unified_database_system");
330 }
331
332 // Connect to database using connection_config
333 auto config = core::connection_config::from_string(connection_string);
334 auto result = backend_->initialize(config);
335 if (result.is_err()) {
336 return make_error("Connection failed: " + result.error().message, static_cast<int>(error_code::connection_failed), "unified_database_system");
337 }
338
339 connected_ = true;
340
341 // Log connection
342 if (auto* logger = coordinator_->get_logger()) {
343 logger->log_connection_event(
344 "connected",
345 "Connection established to: " + connection_string
346 );
347 }
348
349 return kcenon::common::ok();
350 }
351
352 kcenon::common::VoidResult disconnect() {
353 std::lock_guard<std::mutex> lock(mutex_);
354
355 if (!connected_) {
356 return kcenon::common::ok();
357 }
358
359 // Disconnect backend
360 if (backend_) {
361 backend_->shutdown();
362 backend_.reset();
363 }
364
365 connected_ = false;
366
367 // Log disconnection
368 if (auto* logger = coordinator_->get_logger()) {
369 logger->log_connection_event(
370 "disconnected",
371 "Connection closed: " + connection_string_
372 );
373 }
374
375 return kcenon::common::ok();
376 }

379 std::lock_guard<std::mutex> lock(mutex_);

383 // Query execution (sync)
384
385 kcenon::common::Result<query_result> execute(
386 const std::string& query,
387 const std::vector<query_param>& params) {
388
389 std::lock_guard<std::mutex> lock(mutex_);
390
391 if (!connected_) {
392 return make_error_result<query_result>("Not connected to database", static_cast<int>(error_code::invalid_state), "unified_database_system");
393 }
394
395 // Record query start
396 auto start = std::chrono::steady_clock::now();
397
398 // Execute query — use prepared statements when params are provided
399 kcenon::common::Result<core::database_result> db_result = [&]() {
400 if (params.empty()) {
401 if (is_select_query(query)) {
402 return backend_->select_query(query);
403 }
404 auto exec_res = backend_->execute_query(query);
405 if (exec_res.is_err()) {
406 return kcenon::common::Result<core::database_result>(exec_res.error());
407 }
408 return kcenon::common::Result<core::database_result>(core::database_result{});
409 }
410 auto values = convert_params(params);
411 if (is_select_query(query)) {
412 return backend_->select_prepared(query, values);
413 }
414 auto exec_res = backend_->execute_prepared(query, values);
415 if (exec_res.is_err()) {
416 return kcenon::common::Result<core::database_result>(exec_res.error());
417 }
418 return kcenon::common::Result<core::database_result>(core::database_result{});
419 }();
420
421 // Calculate execution time
422 auto duration = std::chrono::duration_cast<std::chrono::microseconds>(
423 std::chrono::steady_clock::now() - start);
424
425 if (db_result.is_err()) {
426 update_metrics(duration, false);
427 return db_result.error();
428 }
429
430 // Update metrics
431 update_metrics(duration, true);
432
433 // Log query
434 if (auto* logger = coordinator_->get_logger()) {
435 logger->log(
436 config_.logger.min_log_level,
437 "Query executed: " + query + " (" + std::to_string(duration.count()) + "us)"
438 );
439 }
440
441 // Record metrics
442 if (auto* monitor = coordinator_->get_monitor()) {
443 monitor->record_query_execution(duration, true);
444 }
445
446 auto qr = convert_result(db_result.value(), duration);
447 // execute_query / execute_prepared return VoidResult (no row count),
448 // so affected_rows stays 0 after convert_result. For successful DML
449 // queries we report at least 1 affected row.
450 if (!is_select_query(query) && qr.affected_rows == 0) {
451 qr.affected_rows = 1;
452 }
453 return qr;
454 }

456 // Query execution (async)
457
458 std::future<kcenon::common::Result<query_result>> execute_async(
459 const std::string& query,
460 const std::vector<query_param>& params) {
461
462 // Submit to thread pool
463 auto* thread_pool = coordinator_->get_thread_pool();
464 if (!thread_pool) {
465 std::promise<kcenon::common::Result<query_result>> promise;
466 promise.set_value(make_error_result<query_result>("Thread pool not available", static_cast<int>(error_code::invalid_state), "unified_database_system"));
467 return promise.get_future();
468 }
469
470 // Create task
471 return thread_pool->submit([this, query, params]() -> kcenon::common::Result<query_result> {
472 return this->execute(query, params);
473 });
474 }
475
476 std::future<kcenon::common::Result<query_result>> execute_async_priority(
477 const std::string& query,
478 int priority,
479 const std::vector<query_param>& params) {
480
481 auto* thread_pool = coordinator_->get_thread_pool();
482 if (!thread_pool) {
483 std::promise<kcenon::common::Result<query_result>> promise;
484 promise.set_value(make_error_result<query_result>("Thread pool not available", static_cast<int>(error_code::invalid_state), "unified_database_system"));
485 return promise.get_future();
486 }
487
488 // Note: Priority parameter is currently ignored
489 // Backend pattern removed priority support for simplification
490 (void)priority; // Suppress unused parameter warning
491
492 return thread_pool->submit([this, query, params]() -> kcenon::common::Result<query_result> {
493 return this->execute(query, params);
494 });
495 }

497 // Transaction management
498
499 kcenon::common::Result<std::unique_ptr<transaction>> begin_transaction() {
500 std::lock_guard<std::mutex> lock(mutex_);
501
502 if (!connected_) {
503 return make_error_result<std::unique_ptr<transaction>>("Not connected to database", static_cast<int>(error_code::invalid_state), "unified_database_system");
504 }
505
506 ++metrics_.transactions_started;
507
508 auto tx_impl = std::make_unique<transaction_impl>(backend_);
509 std::unique_ptr<transaction> tx = std::move(tx_impl);
510 return tx;
511 }
512
513 kcenon::common::VoidResult execute_transaction(const std::vector<std::string>& queries) {
514 auto tx_result = begin_transaction();
515 if (!tx_result.is_ok()) {
516 return tx_result.error();
517 }
518
519 auto tx = std::move(tx_result.value());
520
521 for (const auto& query : queries) {
522 auto result = tx->execute(query, {});
523 if (!result.is_ok()) {
524 tx->rollback();
525 ++metrics_.transactions_rolled_back;
526 return result.error();
527 }
528 }
529
530 auto commit_result = tx->commit();
531 if (commit_result.is_ok()) {
532 ++metrics_.transactions_committed;
533 } else {
534 ++metrics_.transactions_rolled_back;
535 }
536
537 return commit_result;
538 }

628 void update_metrics(std::chrono::microseconds latency, bool success) {
629 ++metrics_.total_queries;
630
631 if (success) {
632 ++metrics_.successful_queries;
633 } else {
634 ++metrics_.failed_queries;
635 }
636
637 // Update latency stats
638 if (metrics_.min_latency == std::chrono::microseconds{0} ||
639 latency < metrics_.min_latency) {
640 metrics_.min_latency = latency;
641 }
642
643 if (latency > metrics_.max_latency) {
644 metrics_.max_latency = latency;
645 }
646
647 // Exponential Moving Average (EMA) with alpha = 0.1 (gives more weight to recent values)
648 // EMA formula: EMA_new = alpha * value + (1 - alpha) * EMA_old
649 constexpr double alpha = 0.1;
650 if (metrics_.total_queries == 1) {
651 // First query: initialize with actual latency
652 metrics_.average_latency = latency;
653 } else {
654 // Subsequent queries: apply EMA formula
655 auto current_avg_us = metrics_.average_latency.count();
656 auto new_latency_us = latency.count();
657 auto ema_us = static_cast<int64_t>(alpha * new_latency_us + (1.0 - alpha) * current_avg_us);
658 metrics_.average_latency = std::chrono::microseconds{ema_us};
659 }
660
661 // Calculate throughput
662 auto duration = std::chrono::steady_clock::now() - metrics_.measurement_start;
663 auto seconds = std::chrono::duration_cast<std::chrono::seconds>(duration).count();
664 if (seconds > 0) {
665 metrics_.queries_per_second = static_cast<double>(metrics_.total_queries) / seconds;
666 }
667
668 // Check for slow query
669 if (latency > config_.logger.slow_query_threshold) {
670 ++metrics_.slow_queries;
671 }
672 }
