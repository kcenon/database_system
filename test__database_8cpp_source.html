124// ---------------------------------------------------------------------------
125// Error-code band tests (issue #600)
126//
127// The database error_code enumerators must live in common_system's reserved
128// database_system band [-599, -500] so that when a backend casts one into the
129// shared kcenon::common::error_info::code, common's error registry attributes
130// it to the "DatabaseSystem" category instead of the common (-1..-99) band.
131// ---------------------------------------------------------------------------
132TEST(DatabaseErrorCodeBandTest, RepresentativeCodeIsInDatabaseBand) {
133 const int code = static_cast<int>(error_code::connection_failed);
134
135 // Inside the reserved database_system band.
136 EXPECT_GE(code, -599);
137 EXPECT_LE(code, -500);
138
139 // common's registry resolves it to the DatabaseSystem category.
140 EXPECT_EQ(kcenon::common::error::get_category_name(code), "DatabaseSystem");
141}
142
143TEST(DatabaseErrorCodeBandTest, AlignedCodesMatchCommonDatabaseSystem) {
144 namespace cds = kcenon::common::error::codes::database_system;
145
146 // Codes with a common equivalent are aligned to the exact common value so the
147 // shared message table returns the correct text.
148 EXPECT_EQ(static_cast<int>(error_code::connection_failed), cds::connection_failed);
149 EXPECT_EQ(static_cast<int>(error_code::query_failed), cds::query_failed);
150 EXPECT_EQ(static_cast<int>(error_code::timeout), cds::query_timeout);
151
152 // connection_failed has a dedicated message in common's table.
153 EXPECT_EQ(
154 kcenon::common::error::get_error_message(
155 static_cast<int>(error_code::connection_failed)),
156 "Database connection failed");
157}

159TEST(DatabaseErrorCodeBandTest, AllNonZeroCodesInBandAndDistinct) {
160 const std::vector<int> codes = {
161 static_cast<int>(error_code::connection_failed),
162 static_cast<int>(error_code::query_failed),
163 static_cast<int>(error_code::timeout),
164 static_cast<int>(error_code::unknown_error),
165 static_cast<int>(error_code::invalid_argument),
166 static_cast<int>(error_code::not_implemented),
167 static_cast<int>(error_code::invalid_state),
168 };
169
170 for (int code : codes) {
171 EXPECT_GE(code, -599) << "code " << code << " below database band";
172 EXPECT_LE(code, -500) << "code " << code << " above database band";
173 EXPECT_EQ(kcenon::common::error::get_category_name(code), "DatabaseSystem")
174 << "code " << code << " not attributed to DatabaseSystem";
175 }
176
177 // All enumerators are distinct (no collisions within the band).
178 std::vector<int> sorted = codes;
179 std::sort(sorted.begin(), sorted.end());
180 EXPECT_EQ(std::adjacent_find(sorted.begin(), sorted.end()), sorted.end())
181 << "duplicate error_code value detected";
182
183 // success stays 0.
184 EXPECT_EQ(static_cast<int>(error_code::success), 0);
185}
