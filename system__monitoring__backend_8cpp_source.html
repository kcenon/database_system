65common::VoidResult system_monitoring_backend::initialize()
66{
67 if (initialized_)
68 {
69 return common::ok();
70 }
71
72 try
73 {
74 // Create performance monitor
75 monitor_ = std::make_unique<kcenon::monitoring::performance_monitor>("database_system");
76
77 // Initialize the monitor
78 auto init_result = monitor_->initialize();
79 if (!init_result.is_ok())
80 {
81 return make_error("Failed to initialize monitoring_system: " +
82 std::string(init_result.error().message));
83 }
84
85 start_time_ = std::chrono::steady_clock::now();
86 initialized_ = true;
87 return common::ok();
88 }
89 catch (const std::exception& e)
90 {
91 return make_error("Exception during initialization: " + std::string(e.what()));
92 }

113 monitor_.reset();

115

118 initialized_ = false;
119
120 return common::ok();
121 }

124 return make_error("Exception during shutdown: " + std::string(e.what()));

167common::Result<metrics_snapshot> system_monitoring_backend::get_metrics()
168{
169 if (!initialized_)
170 {
171 return make_error_result<metrics_snapshot>("Monitoring backend not initialized");
172 }
173
174 std::lock_guard<std::mutex> lock(mutex_);
175
176 try
177 {
178 metrics_snapshot snapshot;
179 snapshot.source_id = "database_system";
180
181 // Connection metrics
182 snapshot.gauges["db.connections.active"] = static_cast<double>(active_connections_);
183 snapshot.gauges["db.connections.idle"] = static_cast<double>(idle_connections_);
184 snapshot.gauges["db.connections.total"] = static_cast<double>(total_connections_);
185 snapshot.gauges["db.connections.usage_percent"] = current_metrics_.connection_usage_percent;
186
187 // Query metrics
188 snapshot.counters["db.queries.total"] = current_metrics_.total_queries;
189 snapshot.counters["db.queries.successful"] = current_metrics_.successful_queries;
190 snapshot.counters["db.queries.failed"] = current_metrics_.failed_queries;
191
192 snapshot.gauges["db.query.avg_latency_us"] = static_cast<double>(current_metrics_.avg_query_latency.count());
193 snapshot.gauges["db.query.min_latency_us"] = static_cast<double>(current_metrics_.min_query_latency.count());
194 snapshot.gauges["db.query.max_latency_us"] = static_cast<double>(current_metrics_.max_query_latency.count());
195 snapshot.gauges["db.query.p95_latency_us"] = static_cast<double>(current_metrics_.p95_query_latency.count());
196 snapshot.gauges["db.query.p99_latency_us"] = static_cast<double>(current_metrics_.p99_query_latency.count());
197 snapshot.gauges["db.query.success_rate"] = current_metrics_.query_success_rate;
198
199 // Transaction metrics
200 snapshot.gauges["db.transactions.active"] = static_cast<double>(active_transactions_);
201 snapshot.counters["db.transactions.committed"] = current_metrics_.committed_transactions;
202 snapshot.counters["db.transactions.rolled_back"] = current_metrics_.rolled_back_transactions;
203 snapshot.gauges["db.transaction.commit_rate"] = current_metrics_.transaction_commit_rate;
204
205 // Throughput metrics
206 snapshot.gauges["db.queries_per_second"] = current_metrics_.queries_per_second;
207 snapshot.gauges["db.transactions_per_second"] = current_metrics_.transactions_per_second;
208
209 return snapshot;
210 }
211 catch (const std::exception& e)
212 {
213 return make_error_result<metrics_snapshot>("Failed to get metrics: " + std::string(e.what()));
214 }

217common::Result<health_check_result> system_monitoring_backend::check_health()
218{
219 if (!initialized_)
220 {
221 return make_error_result<health_check_result>("Monitoring backend not initialized");
222 }
223
224 std::lock_guard<std::mutex> lock(mutex_);
225
226 health_check_result result;
227 result.status = health_status::healthy;
228 result.message = "Database system healthy";
229
230 // Connection pool health check
231 if (current_metrics_.connection_usage_percent > config_.connection_usage_warning_threshold * 100.0)
232 {
233 result.status = health_status::degraded;
234 result.message = "Connection pool usage critical";
235 result.metadata["connection_usage"] =
236 std::to_string(current_metrics_.connection_usage_percent) + "%";
237 }
238
239 // Query latency health check
240 if (current_metrics_.avg_query_latency > config_.query_latency_warning)
241 {
242 result.status = health_status::degraded;
243 result.message = "Query latency critical";
244 result.metadata["avg_latency_us"] = std::to_string(current_metrics_.avg_query_latency.count());
245 }
246
247 // Query success rate health check
248 if (current_metrics_.total_queries > 10 && current_metrics_.query_success_rate < 0.95)
249 {
250 result.status = health_status::unhealthy;
251 result.message = "Query success rate low";
252 result.metadata["success_rate"] = std::to_string(current_metrics_.query_success_rate);
253 }
254
255 return result;
256}

289void system_monitoring_backend::record_query_execution(std::chrono::microseconds duration, bool success)
290{
291 std::lock_guard<std::mutex> lock(mutex_);
292
293 // Update counters
294 current_metrics_.total_queries++;
295 if (success)
296 {
297 current_metrics_.successful_queries++;
298 }
299 else
300 {
301 current_metrics_.failed_queries++;
302 }
303
304 // Calculate success rate
305 if (current_metrics_.total_queries > 0)
306 {
307 current_metrics_.query_success_rate =
308 static_cast<double>(current_metrics_.successful_queries) / current_metrics_.total_queries;
309 }
310
311 // Record in monitoring_system
312 if (monitor_)
313 {
314 auto duration_ns = std::chrono::duration_cast<std::chrono::nanoseconds>(duration);
315 monitor_->get_profiler().record_sample("query_execution", duration_ns, success);
316 }
317
318 // Store latency sample
319 recent_query_latencies_.push_back(duration);
320 if (recent_query_latencies_.size() > max_latency_samples_)
321 {
322 recent_query_latencies_.erase(recent_query_latencies_.begin());
323 }
324
325 // Calculate derived metrics
326 calculate_derived_metrics();
327}

371void system_monitoring_backend::record_transaction_commit()
372{
373 std::lock_guard<std::mutex> lock(mutex_);
374
375 if (active_transactions_ > 0)
376 {
377 active_transactions_--;
378 }
379
380 current_metrics_.active_transactions = active_transactions_;
381 current_metrics_.committed_transactions++;
382
383 // Calculate commit rate
384 auto total_txns = current_metrics_.committed_transactions + current_metrics_.rolled_back_transactions;
385 if (total_txns > 0)
386 {
387 current_metrics_.transaction_commit_rate =
388 static_cast<double>(current_metrics_.committed_transactions) / total_txns;
389 }
390}

413common::Result<database_metrics> system_monitoring_backend::get_database_metrics()
414{
415 if (!initialized_)
416 {
417 return make_error_result<database_metrics>("Monitoring backend not initialized");
418 }
419
420 std::lock_guard<std::mutex> lock(mutex_);
421
422 // Update timestamp
423 current_metrics_.timestamp = std::chrono::system_clock::now();
424
425 // Calculate throughput
426 auto elapsed = std::chrono::steady_clock::now() - start_time_;
427 auto elapsed_seconds = std::chrono::duration_cast<std::chrono::seconds>(elapsed).count();
428
429 if (elapsed_seconds > 0)
430 {
431 current_metrics_.queries_per_second =
432 static_cast<double>(current_metrics_.total_queries) / elapsed_seconds;
433 auto total_txns =
434 current_metrics_.committed_transactions + current_metrics_.rolled_back_transactions;
435 current_metrics_.transactions_per_second = static_cast<double>(total_txns) / elapsed_seconds;
436 }
437
438 return current_metrics_;
439}

531void system_monitoring_backend::calculate_derived_metrics()
532{
533 if (recent_query_latencies_.empty())
534 {
535 return;
536 }
537
538 // Create sorted copy for percentile calculation
539 auto sorted_latencies = recent_query_latencies_;
540 std::sort(sorted_latencies.begin(), sorted_latencies.end());
541
542 // Calculate min/max
543 current_metrics_.min_query_latency = sorted_latencies.front();
544 current_metrics_.max_query_latency = sorted_latencies.back();
545
546 // Calculate average
547 auto total_latency = std::accumulate(
548 sorted_latencies.begin(),
549 sorted_latencies.end(),
550 std::chrono::microseconds(0));
551 current_metrics_.avg_query_latency = total_latency / sorted_latencies.size();
552
553 // Calculate percentiles
554 auto calc_percentile = [&sorted_latencies](double percentile) -> std::chrono::microseconds {
555 if (sorted_latencies.empty())
556 return std::chrono::microseconds(0);
557
558 std::size_t index =
559 static_cast<std::size_t>((percentile / 100.0) * (sorted_latencies.size() - 1));
560 return sorted_latencies[index];
561 };
562
563 current_metrics_.p95_query_latency = calc_percentile(95.0);
564 current_metrics_.p99_query_latency = calc_percentile(99.0);
565}
