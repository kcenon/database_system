40public:
41 explicit impl(const unified_db_config& config)
42 : config_(config)
43 , initialized_(false)
44 , logger_(nullptr)
45 , monitor_(nullptr)
46 , thread_pool_(nullptr)
47 , init_time_()

49 }
50
51 ~impl()
52 {
53 if (initialized_)
54 {
55 shutdown();
56 }
57 }
58
59 common::VoidResult initialize()
60 {
61 if (initialized_)
62 {
63 return make_error("Coordinator already initialized");
64 }
65
66 init_time_ = std::chrono::system_clock::now();
67
68 try
69 {
70 // ═══════════════════════════════════════════════════════════════
71 // Phase 1: Initialize Logger (for observability of other inits)
72 // ═══════════════════════════════════════════════════════════════
73 logger_ = std::make_unique<adapters::logger_adapter>(config_.logger);
74
75 auto logger_result = logger_->initialize();
76 if (!logger_result.is_ok())
77 {
78 logger_.reset();
79 return make_error(
80 "Logger initialization failed: "
81#if defined(USE_COMMON_SYSTEM)
82 + logger_result.error().message
83#else
84 + logger_result.error().message
85#endif
86 );
87 }
88
89 logger_->log(db_log_level::info, "Database coordinator: Logger initialized");
90
91 // ═══════════════════════════════════════════════════════════════
92 // Phase 2: Initialize Monitoring (for metrics collection)
93 // ═══════════════════════════════════════════════════════════════
94 monitor_ = std::make_unique<adapters::monitoring_adapter>(config_.monitoring);
95
96 auto monitor_result = monitor_->initialize();
97 if (!monitor_result.is_ok())
98 {
99 logger_->log(db_log_level::error, "Monitoring initialization failed");
100
101 // Rollback: shutdown logger
102 logger_->shutdown();
103 logger_.reset();
104 monitor_.reset();
105
106 return make_error(
107 "Monitoring initialization failed: "
108#if defined(USE_COMMON_SYSTEM)
109 + monitor_result.error().message
110#else
111 + monitor_result.error().message
112#endif
113 );
114 }
115
116 logger_->log(db_log_level::info, "Database coordinator: Monitoring initialized");
117
118 // ═══════════════════════════════════════════════════════════════
119 // Phase 3: Initialize Thread Pool (for async operations)
120 // ═══════════════════════════════════════════════════════════════
121 thread_pool_ = std::make_unique<adapters::thread_adapter>(config_.thread);
122
123 auto thread_result = thread_pool_->initialize();
124 if (!thread_result.is_ok())
125 {
126 logger_->log(db_log_level::error, "Thread pool initialization failed");
127
128 // Rollback: shutdown monitor and logger
129 monitor_->shutdown();
130 monitor_.reset();
131 logger_->shutdown();
132 logger_.reset();
133 thread_pool_.reset();
134
135 return make_error(
136 "Thread pool initialization failed: "
137#if defined(USE_COMMON_SYSTEM)
138 + thread_result.error().message
139#else
140 + thread_result.error().message
141#endif
142 );
143 }
144
145 logger_->log(db_log_level::info, "Database coordinator: Thread pool initialized");
146
147 // ═══════════════════════════════════════════════════════════════
148 // Initialization Complete
149 // ═══════════════════════════════════════════════════════════════
150 initialized_ = true;
151
152 logger_->log(db_log_level::info,
153 "Database coordinator: All adapters initialized successfully");
154
155 return common::ok();
156 }
157 catch (const std::exception& e)
158 {
159 // Clean up any partially initialized adapters
160 if (thread_pool_)
161 {
162 thread_pool_->shutdown();
163 thread_pool_.reset();
164 }
165 if (monitor_)
166 {
167 monitor_->shutdown();
168 monitor_.reset();
169 }
170 if (logger_)
171 {
172 logger_->log(db_log_level::error,
173 std::string("Exception during initialization: ") + e.what());
174 logger_->shutdown();
175 logger_.reset();
176 }
177
178 return make_error(std::string("Exception during initialization: ") + e.what());
179 }
180 }
181
182 common::VoidResult shutdown()
183 {
184 if (!initialized_)
185 {
186 return common::ok(); // Already shut down
187 }
188
189 try
190 {
191 // Shutdown in REVERSE order of initialization
192 // This ensures dependencies are torn down safely
193
194 // ═══════════════════════════════════════════════════════════════
195 // Phase 1: Shutdown Thread Pool
196 // ═══════════════════════════════════════════════════════════════
197 if (thread_pool_)
198 {
199 if (logger_)
200 {
201 logger_->log(db_log_level::info,
202 "Database coordinator: Shutting down thread pool");
203 }
204
205 auto thread_result = thread_pool_->shutdown();
206 if (!thread_result.is_ok())
207 {
208 if (logger_)
209 {
210 logger_->log(db_log_level::warning,
211 "Thread pool shutdown warning: "
212#if defined(USE_COMMON_SYSTEM)
213 + thread_result.error().message
214#else
215 + thread_result.error().message
216#endif
217 );
218 }
219 }
220
221 thread_pool_.reset();
222 }
223
224 // ═══════════════════════════════════════════════════════════════
225 // Phase 2: Shutdown Monitoring
226 // ═══════════════════════════════════════════════════════════════
227 if (monitor_)
228 {
229 if (logger_)
230 {
231 logger_->log(db_log_level::info,
232 "Database coordinator: Shutting down monitoring");
233 }
234
235 auto monitor_result = monitor_->shutdown();
236 if (!monitor_result.is_ok())
237 {
238 if (logger_)
239 {
240 logger_->log(db_log_level::warning,
241 "Monitoring shutdown warning: "
242#if defined(USE_COMMON_SYSTEM)
243 + monitor_result.error().message
244#else
245 + monitor_result.error().message
246#endif
247 );
248 }
249 }
250
251 monitor_.reset();
252 }
253
254 // ═══════════════════════════════════════════════════════════════
255 // Phase 3: Shutdown Logger (LAST - keep logging until the end)
256 // ═══════════════════════════════════════════════════════════════
257 if (logger_)
258 {
259 logger_->log(db_log_level::info,
260 "Database coordinator: Shutdown complete");
261 logger_->flush(); // Ensure all logs are written
262
263 auto logger_result = logger_->shutdown();
264 // Don't log logger shutdown errors (logger is shutting down!)
265
266 logger_.reset();
267 }
268
269 initialized_ = false;
270
271 return common::ok();
272 }
273 catch (const std::exception& e)
274 {
275 // Best effort cleanup
276 thread_pool_.reset();
277 monitor_.reset();
278 logger_.reset();
279
280 initialized_ = false;
281
282 return make_error(std::string("Exception during shutdown: ") + e.what());
283 }
284 }

306 common::Result<bool> check_health()
307 {
308 if (!initialized_)
309 {
310 return common::Result<bool>(
311 common::error_info{ static_cast<int>(error_code::invalid_state), "Coordinator not initialized", "" }
312 );
313 }
314
315 bool overall_healthy = true;
316
317 // Check logger
318 if (!logger_)
319 {
320 overall_healthy = false;
321 }
322
323 // Check monitoring
324 if (!monitor_)
325 {
326 overall_healthy = false;
327 }
328 else
329 {
330 auto monitor_health = monitor_->check_health();
331 if (!monitor_health.is_ok())
332 {
333 overall_healthy = false;
334 if (logger_)
335 {
336 logger_->log(db_log_level::warning, "Monitoring health check failed");
337 }
338 }
339 }
340
341 // Check thread pool
342 if (!thread_pool_)
343 {
344 overall_healthy = false;
345 }
346 // Thread pool is healthy if it exists
347 // (No statistics method available in thread_adapter yet)
348
349 return common::Result<bool>(overall_healthy);
350 }
351
352 common::Result<database_coordinator::coordinator_stats> get_stats() const
353 {
354 coordinator_stats stats;
355 stats.is_initialized = initialized_;
356 stats.logger_healthy = (logger_ != nullptr);
357 stats.monitoring_healthy = (monitor_ != nullptr);
358 stats.thread_pool_healthy = (thread_pool_ != nullptr);
359 stats.init_time = init_time_;
360
361 if (initialized_)
362 {
363 auto now = std::chrono::system_clock::now();
364 stats.uptime
365 = std::chrono::duration_cast<std::chrono::milliseconds>(now - init_time_);
366 }
367 else
368 {
369 stats.uptime = std::chrono::milliseconds(0);
370 }
371
372 return common::Result<coordinator_stats>(stats);
373 }

375private:
376 unified_db_config config_;
377 bool initialized_;
378
379 // Adapters (in initialization order)
380 std::unique_ptr<adapters::logger_adapter> logger_;
381 std::unique_ptr<adapters::monitoring_adapter> monitor_;
382 std::unique_ptr<adapters::thread_adapter> thread_pool_;
383
384 // Statistics
385 std::chrono::system_clock::time_point init_time_;
386};
