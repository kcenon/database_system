56// Regression test for kcenon/database_system#599.
57// Credential encryption must fail closed when no master key is configured:
58// it must NOT silently fall back to a hardcoded default key (which would
59// "encrypt" credentials under a publicly-known key). With no master key set,
60// store_credentials produces no usable ciphertext, so a subsequent
61// get_credentials cannot round-trip the credential back out.
62TEST_F(SecurityTest, CredentialEncryptionFailsClosedWithoutMasterKey) {
63 credential_manager mgr; // No master key configured.
64
65 security_credentials creds;
66 creds.username = "alice";
67 creds.password_hash = "hashed_secret";
68
69 // store_credentials returns true (entry recorded), but the encrypted blob
70 // is empty because encryption fails closed.
71 mgr.store_credentials("conn-1", creds);
72
73 // The credential must NOT be retrievable: a fail-closed encrypt means there
74 // is nothing to decrypt back into a valid credential.
75 auto retrieved = mgr.get_credentials("conn-1");
76 EXPECT_FALSE(retrieved.has_value())
77 << "Credentials round-tripped without a master key — encryption "
78 "silently used a default key instead of failing closed (#599).";
79}
80
81// With a master key configured, the round-trip succeeds, confirming the
82// fail-closed guard does not change behavior when a key is present.
83TEST_F(SecurityTest, CredentialEncryptionRoundTripsWithMasterKey) {
84 credential_manager mgr;
85 mgr.set_master_key("a-strong-32-byte-master-key-value");
86
87 security_credentials creds;
88 creds.username = "bob";
89 creds.password_hash = "hashed_secret";
90
91 mgr.store_credentials("conn-2", creds);
92
93 auto retrieved = mgr.get_credentials("conn-2");
94 ASSERT_TRUE(retrieved.has_value())
95 << "Round-trip failed with a configured master key.";
96 EXPECT_EQ(retrieved->username, "bob");
97 EXPECT_EQ(retrieved->password_hash, "hashed_secret");
98}

100// Field-level encryption must also fail closed: with neither a field key nor a
