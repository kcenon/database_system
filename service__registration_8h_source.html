79inline common::VoidResult register_database_services(
80 common::di::IServiceContainer& container,
81 const database_registration_config& config = {}) {
82
83 // Check if already registered
84 if (container.is_registered<common::interfaces::IDatabase>()) {
85 return common::make_error<std::monostate>(
86 common::di::di_error_codes::already_registered,
87 "IDatabase is already registered",
88 "database_system::di"
89 );
90 }
91
92 // Register database factory
93 return container.register_factory<common::interfaces::IDatabase>(
94 [config](common::di::IServiceContainer&) -> std::shared_ptr<common::interfaces::IDatabase> {
95 // Create adapter with configured database type
96 auto adapter = std::make_shared<adapters::common_system_database_adapter>(config.db_type);
97
98 // Connect if connection string provided and connect_on_register is true
99 if (config.connect_on_register && !config.connection_string.empty()) {
100 auto result = adapter->connect(config.connection_string);
101 if (result.is_err()) {
102 // Return adapter anyway - caller can check is_connected()
103 // This is intentional to avoid throwing in factory
104 }
105 }
106
107 return adapter;
108 },
109 config.lifetime
110 );
111}

134inline common::VoidResult register_database_instance(
135 common::di::IServiceContainer& container,
136 std::shared_ptr<::database::database_manager> manager) {
137
138 if (!manager) {
139 return common::make_error<std::monostate>(
140 common::error_codes::INVALID_ARGUMENT,
141 "Cannot register null database_manager instance",
142 "database_system::di"
143 );
144 }
145
146 auto adapter = std::make_shared<adapters::common_system_database_adapter>(std::move(manager));
147
148 return container.register_instance<common::interfaces::IDatabase>(adapter);
149}
