27std::unique_ptr<backends::monitoring_backend> monitoring_adapter::create_backend(
28 const db_monitoring_config& config,
29 monitoring_backend_type backend_type)
30{
31 switch (backend_type)
32 {
33 case monitoring_backend_type::auto_select:
34 // Prefer system_monitoring_backend if available
35#ifdef HAVE_SYSTEM_MONITORING_BACKEND
36 return std::make_unique<backends::system_monitoring_backend>(config);
37#else
38 return std::make_unique<backends::fallback_monitoring_backend>(config);
39#endif
40
41 case monitoring_backend_type::system:
42#ifdef HAVE_SYSTEM_MONITORING_BACKEND
43 return std::make_unique<backends::system_monitoring_backend>(config);
44#else
45 // Fallback to fallback backend if system backend not available
46 return std::make_unique<backends::fallback_monitoring_backend>(config);
47#endif
48
49 case monitoring_backend_type::fallback:
50 return std::make_unique<backends::fallback_monitoring_backend>(config);
51
52 case monitoring_backend_type::null:
53 return std::make_unique<backends::null_monitoring_backend>(config);
54
55 default:
56 return std::make_unique<backends::fallback_monitoring_backend>(config);
57 }
58}
