107common::Result<metrics_snapshot> fallback_monitoring_backend::get_metrics()
108{
109 if (!initialized_)
110 {
111 return make_error_result<metrics_snapshot>("Monitoring backend not initialized");
112 }
113
114 std::lock_guard<std::mutex> lock(mutex_);
115
116 metrics_snapshot snapshot;
117 snapshot.source_id = "database_fallback";
118
119 // Connection metrics
120 snapshot.gauges["db.connections.active"] = static_cast<double>(metrics_.active_connections);
121 snapshot.gauges["db.connections.idle"] = static_cast<double>(metrics_.idle_connections);
122 snapshot.gauges["db.connections.usage_percent"] = metrics_.connection_usage_percent;
123
124 // Query metrics
125 snapshot.gauges["db.query.avg_latency_us"] = static_cast<double>(metrics_.avg_query_latency.count());
126 snapshot.gauges["db.query.success_rate"] = metrics_.query_success_rate;
127
128 snapshot.counters["db.queries.total"] = metrics_.total_queries;
129 snapshot.counters["db.queries.successful"] = metrics_.successful_queries;
130 snapshot.counters["db.queries.failed"] = metrics_.failed_queries;
131
132 // Transaction metrics
133 snapshot.counters["db.transactions.committed"] = metrics_.committed_transactions;
134 snapshot.counters["db.transactions.rolled_back"] = metrics_.rolled_back_transactions;
135
136 // Generic metrics
137 for (const auto& [name, value] : generic_metrics_)
138 {
139 snapshot.gauges[name] = value;
140 }
141
142 return snapshot;
143}

145common::Result<health_check_result> fallback_monitoring_backend::check_health()
146{
147 if (!initialized_)
148 {
149 return make_error_result<health_check_result>("Monitoring backend not initialized");
150 }
151
152 std::lock_guard<std::mutex> lock(mutex_);
153
154 health_check_result result;
155 result.status = health_status::healthy;
156 result.message = "Database system healthy";
157
158 // Connection pool health check
159 if (metrics_.connection_usage_percent > config_.connection_usage_warning_threshold * 100.0)
160 {
161 result.status = health_status::degraded;
162 result.message = "Connection pool usage critical";
163 result.metadata["connection_usage"] = std::to_string(metrics_.connection_usage_percent) + "%";
164 }
165
166 // Query latency health check
167 if (metrics_.avg_query_latency > config_.query_latency_warning)
168 {
169 result.status = health_status::degraded;
170 result.message = "Query latency critical";
171 result.metadata["avg_latency_us"] = std::to_string(metrics_.avg_query_latency.count());
172 }
173
174 // Query success rate health check
175 if (metrics_.total_queries > 10 && metrics_.query_success_rate < 0.95)
176 {
177 result.status = health_status::degraded;
178 result.message = "Query success rate low";
179 result.metadata["success_rate"] = std::to_string(metrics_.query_success_rate);
180 }
181
182 return result;
183}

199void fallback_monitoring_backend::record_query_execution(std::chrono::microseconds duration, bool success)
200{
201 std::lock_guard<std::mutex> lock(mutex_);
202
203 metrics_.total_queries++;
204 if (success)
205 {
206 metrics_.successful_queries++;
207 }
208 else
209 {
210 metrics_.failed_queries++;
211 }
212
213 if (metrics_.total_queries > 0)
214 {
215 metrics_.query_success_rate = static_cast<double>(metrics_.successful_queries) / metrics_.total_queries;
216 }
217
218 query_latencies_.push_back(duration);
219
220 // Keep last 1000 samples
221 const size_t max_samples = 1000;
222 if (query_latencies_.size() > max_samples)
223 {
224 query_latencies_.erase(query_latencies_.begin());
225 }
226
227 update_avg_latency();
228}

315std::chrono::microseconds fallback_monitoring_backend::calculate_percentile(double percentile)
316{
317 if (query_latencies_.empty())
318 {
319 return std::chrono::microseconds(0);
320 }
321
322 auto sorted = query_latencies_;
323 std::sort(sorted.begin(), sorted.end());
324
325 size_t idx = static_cast<size_t>(sorted.size() * percentile);
326 if (idx >= sorted.size())
327 {
328 idx = sorted.size() - 1;
329 }
330
331 return sorted[idx];
332}

334void fallback_monitoring_backend::update_avg_latency()
335{
336 if (query_latencies_.empty())
337 {
338 return;
339 }
340
341 auto sum = std::accumulate(
342 query_latencies_.begin(), query_latencies_.end(), std::chrono::microseconds(0));
343 metrics_.avg_query_latency = std::chrono::microseconds(sum.count() / query_latencies_.size());
344
345 metrics_.min_query_latency = *std::min_element(query_latencies_.begin(), query_latencies_.end());
346 metrics_.max_query_latency = *std::max_element(query_latencies_.begin(), query_latencies_.end());
347
348 auto sorted = query_latencies_;
349 std::sort(sorted.begin(), sorted.end());
350
351 size_t p95_idx = static_cast<size_t>(sorted.size() * 0.95);
352 size_t p99_idx = static_cast<size_t>(sorted.size() * 0.99);
353
354 if (p95_idx < sorted.size())
355 {
356 metrics_.p95_query_latency = sorted[p95_idx];
357 }
358 if (p99_idx < sorted.size())
359 {
360 metrics_.p99_query_latency = sorted[p99_idx];
361 }
362}
