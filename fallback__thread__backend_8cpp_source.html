48common::VoidResult fallback_thread_backend::initialize()
49{
50 if (initialized_)
51 {
52 return common::ok();
53 }
54
55 try
56 {
57 shutdown_requested_ = false;
58 active_tasks_ = 0;
59
60 // Determine worker count
61 std::size_t worker_count = config_.thread_count;
62 if (worker_count == 0)
63 {
64 worker_count = std::max(1u, std::thread::hardware_concurrency());
65 }
66
67 // Create worker threads
68 workers_.reserve(worker_count);
69 for (std::size_t i = 0; i < worker_count; ++i)
70 {
71 workers_.emplace_back(&fallback_thread_backend::worker_thread, this);
72 }
73
74 initialized_ = true;
75 return common::ok();
76 }
77 catch (const std::exception& e)
78 {
79 return make_error(std::string("Thread pool initialization failed: ") + e.what());
80 }
81}

83common::VoidResult fallback_thread_backend::shutdown()
84{
85 if (!initialized_)
86 {
87 return common::ok();
88 }
89
90 try
91 {
92 // Signal shutdown
93 shutdown_requested_ = true;
94
95 // Wake up all workers
96 {
97 std::lock_guard<std::mutex> lock(queue_mutex_);
98 queue_cv_.notify_all();
99 }
100
101 // Wait for all workers to finish
102 for (auto& worker : workers_)
103 {
104 if (worker.joinable())
105 {
106 worker.join();
107 }
108 }
109
110 workers_.clear();
111
112 // Clear remaining tasks
113 {
114 std::lock_guard<std::mutex> lock(queue_mutex_);
115 std::queue<std::function<void()>> empty;
116 task_queue_.swap(empty);
117 }
118
119 initialized_ = false;
120 return common::ok();
121 }
122 catch (const std::exception& e)
123 {
124 return make_error(std::string("Thread pool shutdown failed: ") + e.what());
125 }
126}

133common::VoidResult fallback_thread_backend::execute(std::function<void()> task)
134{
135 if (!initialized_)
136 {
137 return make_error("Thread pool not initialized");
138 }
139
140 if (!task)
141 {
142 return make_error("Invalid task");
143 }
144
145 if (shutdown_requested_)
146 {
147 return make_error("Thread pool is shutting down");
148 }
149
150 {
151 std::lock_guard<std::mutex> lock(queue_mutex_);
152
153 // Check queue size limit
154 if (config_.max_queue_size > 0 && task_queue_.size() >= config_.max_queue_size)
155 {
156 return make_error("Task queue full");
157 }
158
159 task_queue_.push(std::move(task));
160 }
161
162 queue_cv_.notify_one();
163 return common::ok();
164}

211void fallback_thread_backend::worker_thread()
212{
213 while (true)
214 {
215 std::function<void()> task;
216
217 {
218 std::unique_lock<std::mutex> lock(queue_mutex_);
219
220 // Wait for task or shutdown
221 queue_cv_.wait(lock, [this]() {
222 return shutdown_requested_ || !task_queue_.empty();
223 });
224
225 // Check shutdown
226 if (shutdown_requested_ && task_queue_.empty())
227 {
228 break;
229 }
230
231 // Get next task
232 if (!task_queue_.empty())
233 {
234 task = std::move(task_queue_.front());
235 task_queue_.pop();
236 }
237 }
238
239 // Execute task
240 if (task)
241 {
242 ++active_tasks_;
243
244 try
245 {
246 task();
247 }
248 catch (...)
249 {
250 // Swallow exceptions to prevent worker thread from terminating
251 }
252
253 --active_tasks_;
254
255 // Notify completion waiters
256 {
257 std::lock_guard<std::mutex> lock(completion_mutex_);
258 completion_cv_.notify_all();
259 }
260 }
261 }
