32 std::string format_timestamp()
33 {
34 auto now = std::chrono::system_clock::now();
35 auto time_t_now = std::chrono::system_clock::to_time_t(now);
36 auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(now.time_since_epoch()) % 1000;
37
38 std::stringstream ss;
39 ss << std::put_time(std::localtime(&time_t_now), "%Y-%m-%d %H:%M:%S");
40 ss << '.' << std::setfill('0') << std::setw(3) << ms.count();
41 return ss.str();
42 }

95common::VoidResult fallback_logger_backend::initialize()
96{
97 if (initialized_)
98 {
99 return common::ok();
100 }
101
102 try
103 {
104 // Open log file if file logging enabled
105 if (config_.enable_file_logging)
106 {
107 // Create directory if it doesn't exist
108 std::filesystem::path log_dir(config_.log_directory);
109 if (!std::filesystem::exists(log_dir))
110 {
111 std::filesystem::create_directories(log_dir);
112 }
113
114 std::string log_path = config_.log_directory + "/database.log";
115 log_file_.open(log_path, std::ios::app);
116 if (!log_file_.is_open())
117 {
118 return make_error("Failed to open log file: " + log_path);
119 }
120 }
121
122 initialized_ = true;
123 return common::ok();
124 }
125 catch (const std::exception& e)
126 {
127 return make_error(std::string("Logger initialization failed: ") + e.what());
128 }
129}

131common::VoidResult fallback_logger_backend::shutdown()
132{
133 if (!initialized_)
134 {
135 return common::ok();
136 }
137
138 try
139 {
140 std::lock_guard<std::mutex> lock(mutex_);

143 log_file_.flush();
144 log_file_.close();
145 }
146 initialized_ = false;
147 return common::ok();
148 }
149 catch (const std::exception& e)
150 {
151 return make_error(std::string("Logger shutdown failed: ") + e.what());
152 }

160void fallback_logger_backend::log(db_log_level level, const std::string& message)
161{
162 if (!initialized_)
163 {
164 return;
165 }
166
167 // Check if this level should be logged
168 if (level < config_.min_log_level)
169 {
170 return;
171 }
172
173 std::lock_guard<std::mutex> lock(mutex_);
174
175 // Format: [2025-01-03 14:30:45.123] [INFO ] message
176 std::string log_line
177 = "[" + format_timestamp() + "] [" + log_level_to_string(level) + "] " + message;
178
179 // Write to console
180 std::cout << log_line << std::endl;
181
182 // Write to file if enabled
183 if (config_.enable_file_logging && log_file_.is_open())
184 {
185 log_file_ << log_line << std::endl;
186 }
187}
